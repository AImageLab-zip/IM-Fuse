\begin{figure}[h]

  colback=lightgreenbg,

  boxrule=0.6pt,
  title=MiMoSe preprocess,
  fonttitle=\bfseries
]
\begin{Verbatim}[breaklines=true,breakanywhere=true,fontsize=\small]
Options
  [Config / I/O]
  --config PATH                         Config
  --input-dir DIRECTORY                 Input dir
  --output-dir DIRECTORY                Output dir
  --dataset-type [brats18|brats23]      Dataset

  [Cropping]
  --crop-mode TEXT                      Crop mode
  --crop-size INT INT INT               Crop size
  --crop-min-size INT INT INT           Min crop

  [Clamping]
  --clamp-mode TEXT                     Clamp mode
  --clamp-percentile FP [FP]            Percentiles
  --clamp-min INT [INT ...]             Clamp min
  --clamp-max INT [INT ...]             Clamp max

  [Normalization]
  --norm-mode TEXT                      Norm mode
  --norm-min-max-range FP FP            Min-max range
  --norm-mean FP FP FP FP               Mean
  --norm-std FP FP FP FP                Std

  [Execution]
  --yes, -y                             Confirm
  --help                                Help
\end{Verbatim}
\end{tcolorbox}
\caption{Help box for the MiMoSe preprocess command. FP means floating-point number, INT means integer number, DIR means existing directory name, and STR means string.}
\label{fig:mimose_preprocess}
\end{figure}
